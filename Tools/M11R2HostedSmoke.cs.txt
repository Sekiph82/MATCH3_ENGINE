// CI-only hosted real-network test. This source is copied into a disposable
// Unity checkout; it is never included in a shipping Engine package or game.
using System;
using System.Security.Cryptography;
using System.Text;
using System.Threading;
using System.Threading.Tasks;
using Match3.Engine.Runtime;
using NUnit.Framework;
using UnityEngine;

namespace Match3.Engine.HostedOptionalAddressables
{
    public sealed class M11HostedR2ReadOnlyTests
    {
        // Public, tiny, Engine-neutral fixture. The private signing key is
        // EPHEMERAL in the hosted runner; never shipped or uploaded to R2.
        private const string ExpectedBody =
            "{\"neutral\":\"match3-r2-qa-v1\",\"level\":1}";
        private const string ExpectedDigest =
            "0160ce606a68def670f4f74600c6a6c08fd91dae2ad9e2caedab3e827de42e53";
        private const string Origin =
            "https://pub-39632e9ddf304be7b407919f185cceac.r2.dev/";
        private const string FixturePath = "qa/neutral-v2.json";

        [Test]
        public async Task M11_003_CloudflareR2PublicReadUsesPinnedSignedManifest()
        {
            var expected = Encoding.UTF8.GetBytes(ExpectedBody);
            Assert.That(expected.Length, Is.EqualTo(39));
            Assert.That(RemoteContentSafety.Sha256(expected),
                Is.EqualTo(ExpectedDigest));
            using (var signer = new RSACryptoServiceProvider(2048))
            {
                var manifest = new RemoteContentManifest(
                    "Coral_Rescue", RemotePlatform.Android,
                    RemoteChannel.FamilyTest, "r2-read-only-qa-v1",
                    "test-public-key", new EngineVersion(1, 0, 0),
                    new EngineVersion(1, 1, 0), 1000, 9000,
                    new[] {
                        new RemoteContentEntry(FixturePath,
                            RemoteContentKind.Json, expected.Length,
                            ExpectedDigest)
                    });
                var k = signer.ExportParameters(false);
                var trust = new RemotePublicSigningKey("test-public-key",
                    k.Modulus, k.Exponent, 0, 10000);
                var signed = signer.SignData(manifest.CanonicalBytes(),
                    HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
                var entry = manifest.Entries[0];
                var receipt = await RemoteDownloadCoordinator.FetchVerifiedAsync(
                    manifest, signed, new[] { trust },
                    new RemoteReadOrigin(Origin),
                    new RemoteHttpsReadTransport(),
                    entry, manifest.GameId, manifest.Platform,
                    manifest.Channel, new EngineVersion(1, 0, 0),
                    2000, CancellationToken.None, maxAttempts: 2);
                Assert.That(receipt.Bytes, Is.EqualTo(expected));
                Assert.That(receipt.Bytes.Length, Is.EqualTo(39));
                Assert.That(receipt.Path, Is.EqualTo(FixturePath));
                Assert.That(receipt.Attempts, Is.InRange(1, 2));
                Assert.That(RemoteContentSafety.Sha256(receipt.Bytes),
                    Is.EqualTo(ExpectedDigest));
            }
        }

        // This second test does NOT generate a new signature for the remote
        // object. It verifies the canonical manifest and signature fetched
        // from real Cloudflare R2 against a code-pinned QA public key.
        private const string PinnedQaModulusBase64 =
            "rEhpHM6pi7K9QJWueacRKZkTVrhOBkXLMq9aHjfCaCYFUjC9BfVhbDNpxi4WDxxfEPgjdJP4b3JFbwcka8yKE5qYIeLPuLEg6nmIuYTAZ3W/YJlbqnZ+I+3WJvGroQkPtyRg/uI9WbSoXa7e7uh0m4crqsfbnrN6eMbiShO8BKFgH1KlpCzNjgoR+XjEa2gFv+Vsjx2bImxKwUtMp2fjqbnkoNpXcA8/gc1kidHn0nDgBlV0EPD1EZmgxEZ1Ntw74cVndJQUbI6grZwOePa/zMCrzvCa7W+nsLndc2rZKMstBwU4Lk892bY+IdBTKV5kKKYuib2IJM5TxfttOXZfEQ==";
        private const string PinnedCanonicalSha256 =
            "0af60d84ddf9a140008b4dc2d751bbe027a0c177c5d4b97e416eb5e92e6b15ac";

        [Serializable]
        private sealed class QaSignedEnvelope
        {
            public int schema;
            public string manifest_b64;
            public string signature_b64;
            public string key_id;
        }

        [Test]
        public async Task M11_003_R2RemoteSignedManifestAndPayloadValidateWithPinnedPublicKey()
        {
            var transport = new RemoteHttpsReadTransport();
            var envelopeResponse = await transport.FetchAsync(
                new Uri(Origin + "qa/signed-manifest-v1.json"),
                2048, CancellationToken.None);
            Assert.That(envelopeResponse.HttpStatus, Is.EqualTo(200));
            var json = Encoding.UTF8.GetString(envelopeResponse.Bytes);
            var envelope = JsonUtility.FromJson<QaSignedEnvelope>(json);
            Assert.That(envelope, Is.Not.Null);
            Assert.That(envelope.schema, Is.EqualTo(1));
            Assert.That(envelope.key_id, Is.EqualTo("match3-qa-pub-v1"));
            var wire = Convert.FromBase64String(envelope.manifest_b64);
            var signature = Convert.FromBase64String(envelope.signature_b64);
            Assert.That(wire.Length, Is.EqualTo(196));
            Assert.That(RemoteContentSafety.Sha256(wire),
                Is.EqualTo(PinnedCanonicalSha256));
            var qaKey = new RemotePublicSigningKey("match3-qa-pub-v1",
                Convert.FromBase64String(PinnedQaModulusBase64),
                Convert.FromBase64String("AQAB"), 0, 1823472000);
            const long qaNow = 1792100000;
            var manifest = RemoteManifestCodec.ParseVerified(
                wire, signature, new[] { qaKey }, "Coral_Rescue",
                RemotePlatform.Android, RemoteChannel.FamilyTest,
                new EngineVersion(1, 0, 0), qaNow);
            Assert.That(manifest.ReleaseId, Is.EqualTo("match3-qa-v1"));
            Assert.That(manifest.Entries.Count, Is.EqualTo(1));
            Assert.That(manifest.Entries[0].Sha256,
                Is.EqualTo(ExpectedDigest));
            var downloaded = await RemoteDownloadCoordinator.FetchVerifiedAsync(
                manifest, signature, new[] { qaKey },
                new RemoteReadOrigin(Origin), transport, manifest.Entries[0],
                manifest.GameId, manifest.Platform, manifest.Channel,
                new EngineVersion(1, 0, 0), qaNow, CancellationToken.None);
            Assert.That(downloaded.Bytes,
                Is.EqualTo(Encoding.UTF8.GetBytes(ExpectedBody)));
            var invalid = (byte[])signature.Clone();
            invalid[0] ^= 1;
            Assert.Throws<InvalidOperationException>(() =>
                RemoteManifestCodec.ParseVerified(wire, invalid,
                    new[] { qaKey }, manifest.GameId, manifest.Platform,
                    manifest.Channel, new EngineVersion(1, 0, 0), qaNow));
        }
    }
}
